rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    match /users/{userId}/histories/{historyId} {

      function isOwner() {
        return request.auth != null && request.auth.uid == userId;
      }

      function validCalculation() {
        let d = request.resource.data;
        return d.keys().hasAll([
                 'hargaAwal', 'diskon1', 'diskon2', 'hargaSetelahDiskon1',
                 'hargaAkhir', 'totalHemat', 'persenHemat', 'createdAt'
               ])
          && d.keys().hasOnly([
                 'hargaAwal', 'diskon1', 'diskon2', 'hargaSetelahDiskon1',
                 'hargaAkhir', 'totalHemat', 'persenHemat', 'createdAt'
               ])
          && d.hargaAwal is number && d.hargaAwal > 0
          && d.diskon1 is number && d.diskon1 >= 0 && d.diskon1 <= 100
          && d.diskon2 is number && d.diskon2 >= 0 && d.diskon2 <= 100
          && d.hargaSetelahDiskon1 is number && d.hargaSetelahDiskon1 >= 0
          && d.hargaAkhir is number && d.hargaAkhir >= 0
          && d.totalHemat is number && d.totalHemat >= 0
          && d.persenHemat is number && d.persenHemat >= 0 && d.persenHemat <= 100
          && d.createdAt is timestamp && d.createdAt == request.time;
      }

      allow create: if isOwner() && validCalculation();
      allow get, list, delete: if isOwner();
      allow update: if false;
    }
  }
}
